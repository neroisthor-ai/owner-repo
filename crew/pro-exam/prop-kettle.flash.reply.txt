export function stovetop_kettle({ color = "#b83a2e" } = {}) {
  const g = new THREE.Group();
  const mBody = mat(color, { roughness: 0.28, metalness: 0.12 });
  const mSteel = mat("#d8dcde", { roughness: 0.2, metalness: 0.85 });
  const mBlack = mat("#1e1e1e", { roughness: 0.45, metalness: 0.1 });
  const cx = -0.035;
  cyl(g, 0.088, 0.088, 0.003, mSteel, cx, 0, 0, 32);
  const bodyPoints = [
    new THREE.Vector2(0, 0), new THREE.Vector2(0.086, 0), new THREE.Vector2(0.093, 0.007), new THREE.Vector2(0.095, 0.025),
    new THREE.Vector2(0.095, 0.055), new THREE.Vector2(0.091, 0.085), new THREE.Vector2(0.082, 0.112), new THREE.Vector2(0.070, 0.128),
    new THREE.Vector2(0.055, 0.130), new THREE.Vector2(0.052, 0.124), new THREE.Vector2(0, 0.124)
  ];
  const bodyGeom = new THREE.LatheGeometry(bodyPoints, 32);
  mesh(bodyGeom, mBody, g, cx, 0, 0);
  cyl(g, 0.056, 0.056, 0.004, mSteel, cx, 0.126, 0, 32);
  const lidPoints = [
    new THREE.Vector2(0, 0.018), new THREE.Vector2(0.018, 0.017), new THREE.Vector2(0.035, 0.012),
    new THREE.Vector2(0.048, 0.005), new THREE.Vector2(0.054, 0.001), new THREE.Vector2(0.054, 0)
  ];
  const lidGeom = new THREE.LatheGeometry(lidPoints, 32);
  mesh(lidGeom, mBody, g, cx, 0.128, 0);
  cyl(g, 0.015, 0.015, 0.002, mSteel, cx, 0.146, 0, 20);
  cyl(g, 0.008, 0.012, 0.008, mBlack, cx, 0.148, 0, 16);
  const knobHead = mesh(new THREE.SphereGeometry(0.013, 20, 14), mBlack, g, cx, 0.160, 0);
  knobHead.scale.set(1, 0.65, 1);
  const spoutAngle = 0.70;
  const spoutG = new THREE.Group();
  spoutG.position.set(0.076, 0.108, 0);
  spoutG.rotation.z = -spoutAngle;
  g.add(spoutG);
  const spoutLen = 0.096;
  mesh(new THREE.CylinderGeometry(0.014, 0.024, spoutLen, 24), mBody, spoutG, 0, 0, 0);
  const collarY = spoutLen / 2 + 0.008;
  mesh(new THREE.CylinderGeometry(0.0142, 0.0145, 0.016, 24), mSteel, spoutG, 0, collarY, 0);
  const capY = spoutLen / 2 + 0.017;
  mesh(new THREE.CylinderGeometry(0.0148, 0.0148, 0.004, 24), mSteel, spoutG, 0, capY, 0);
  mesh(new THREE.CylinderGeometry(0.004, 0.004, 0.001, 16), mBlack, spoutG, 0, capY + 0.0021, 0);
  box(spoutG, 0.008, 0.014, 0.006, mBlack, -0.004, capY - 0.006, -0.003);
  box(g, 0.016, 0.012, 0.008, mSteel, cx - 0.008, 0.116, 0.072);
  box(g, 0.016, 0.012, 0.008, mSteel, cx - 0.008, 0.116, -0.080);
  const pinFront = mesh(new THREE.CylinderGeometry(0.0035, 0.0035, 0.020, 12), mSteel, g, cx, 0.124, 0.076);
  pinFront.rotation.z = Math.PI / 2;
  const pinBack = mesh(new THREE.CylinderGeometry(0.0035, 0.0035, 0.020, 12), mSteel, g, cx, 0.124, -0.076);
  pinBack.rotation.z = Math.PI / 2;
  mesh(new THREE.CylinderGeometry(0.0105, 0.0105, 0.008, 16), mSteel, g, cx, 0.136, 0.075);
  mesh(new THREE.CylinderGeometry(0.0105, 0.0105, 0.008, 16), mSteel, g, cx, 0.136, -0.075);
  const handlePoints = [
    new THREE.Vector3(cx, 0.126, 0.075), new THREE.Vector3(cx, 0.168, 0.073), new THREE.Vector3(cx, 0.208, 0.055),
    new THREE.Vector3(cx, 0.228, 0.028), new THREE.Vector3(cx, 0.231, 0), new THREE.Vector3(cx, 0.228, -0.028),
    new THREE.Vector3(cx, 0.208, -0.055), new THREE.Vector3(cx, 0.168, -0.073), new THREE.Vector3(cx, 0.126, -0.075)
  ];
  const handleCurve = new THREE.CatmullRomCurve3(handlePoints);
  const handleGeom = new THREE.TubeGeometry(handleCurve, 36, 0.009, 14, false);
  handleGeom.computeBoundingBox();
  const handleMaxY = handleGeom.boundingBox.max.y;
  handleGeom.translate(0, 0.240 - handleMaxY, 0);
  mesh(handleGeom, mBlack, g, 0, 0, 0);
  const box3 = new THREE.Box3().setFromObject(g);
  const shiftX = (box3.min.x + box3.max.x) / 2;
  const shiftZ = (box3.min.z + box3.max.z) / 2;
  for (const child of g.children) {
    child.position.x -= shiftX;
    child.position.z -= shiftZ;
  }
  return finish(g, "stovetop_kettle", { onSurface: true });
}
